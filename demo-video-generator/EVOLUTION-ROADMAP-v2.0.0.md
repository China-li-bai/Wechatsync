class FeatureConsistency:
    def check_consistency(self, frames: List) -> float:
        """检查特征一致性"""
        features = [self._extract_features(f) for f in frames]
        
        # 计算相似度
        similarities = []
        for i in range(len(features) - 1):
            sim = self._calculate_similarity(features[i], features[i+1])

        return sum(similarities) / len(similarities)
